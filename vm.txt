# VLR Analyst - VM100 one-time SSH setup. Run in an ADMIN PowerShell:
#   irm https://brandon0403.github.io/vlr-analyst/vm.txt | iex
$ErrorActionPreference = 'Continue'
Write-Host "[1/5] Installing OpenSSH Server (needs internet, may take a minute)..."
Add-WindowsCapability -Online -Name OpenSSH.Server~~~~0.0.1.0 | Out-Null
Write-Host "[2/5] Starting sshd..."
Set-Service -Name sshd -StartupType Automatic
Start-Service sshd
Write-Host "[3/5] Firewall rule..."
if (-not (Get-NetFirewallRule -Name sshd -ErrorAction SilentlyContinue)) {
  New-NetFirewallRule -Name sshd -DisplayName 'OpenSSH Server' -Enabled True -Direction Inbound -Protocol TCP -Action Allow -LocalPort 22 | Out-Null
}
Write-Host "[4/5] Registering key..."
$k = 'ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIO1JiSfukZ1KSilIwXkE/naH/3TRfsHCBOKq2TP2Z/nz claude-code@kjh0403-wp3'
$adm = 'C:\ProgramData\ssh\administrators_authorized_keys'
if (-not (Test-Path $adm) -or -not (Select-String -Path $adm -SimpleMatch $k -Quiet)) { Add-Content -Path $adm -Value $k }
icacls $adm /inheritance:r /grant "Administrators:F" /grant "SYSTEM:F" | Out-Null
$uk = Join-Path $env:USERPROFILE '.ssh\authorized_keys'
New-Item -ItemType Directory -Force (Split-Path $uk) | Out-Null
if (-not (Test-Path $uk) -or -not (Select-String -Path $uk -SimpleMatch $k -Quiet)) { Add-Content -Path $uk -Value $k }
Restart-Service sshd
Write-Host "[5/5] Done. Tell Claude these lines:"
"  user     : $env:USERNAME"
"  computer : $env:COMPUTERNAME"
Get-NetIPAddress -AddressFamily IPv4 | Where-Object { $_.IPAddress -like '192.168.*' } | ForEach-Object { "  ip       : " + $_.IPAddress }
"  sshd     : " + (Get-Service sshd).Status
$py = 'none'; try { $py = (& python --version 2>&1) } catch {}
"  python   : $py"
